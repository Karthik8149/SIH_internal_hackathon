%% SIH26058 - 3-Sensor Adaptive LFM Visualizer
% Arduino -> MATLAB
%
% THREE INPUTS:
%   1) Temperature potentiometer
%   2) Depth potentiometer (simulated environmental depth)
%   3) HC-SR04 physical target range
%
% MATLAB receives:
%   DATA,T_C,Depth_m,Range_cm,fc_kHz,BW_kHz,Mode
%
% IMPORTANT:
%   - Frequency axis is fixed at 0-550 kHz.
%   - Center frequency is adaptive.
%   - Bandwidth is CONSTANT at 50 kHz.
%   - HC-SR04 is a hardware target-range demonstrator, not an
%     underwater depth sensor.

clear;
clc;
close all;

%% ====================== USER SETTING ===============================
% CHANGE ONLY THIS IF YOUR ARDUINO USES ANOTHER COM PORT
port = "COM3";

baud = 115200;

%% ====================== SERIAL CONNECTION ==========================
try
    s = serialport(port, baud);
    configureTerminator(s, "LF");
    flush(s);
catch ME
    error(['Could not open %s.\n' ...
           'Check Arduino Tools > Port, close Arduino Serial Monitor, ' ...
           'and make sure the Arduino is connected.\n\nMATLAB error: %s'], ...
           port, ME.message);
end

pause(2);

%% ====================== LFM SETTINGS ===============================
Fs = 2e6;
Tchirp = 1e-3;
N = round(Fs*Tchirp);

t = (0:N-1)/Fs;

% Initial values
Tenv = 20;
Depth_m = 15;
Range_cm = 40;

fc_kHz = 200;
BW_kHz = 50;
Mode = "C3";

%% ====================== MODE REFERENCE DATA ========================
modeNames = ["C1","C2","C3","C4","C5"];
modeFc = [50 100 200 350 500];

bandLow = modeFc - 25;
bandHigh = modeFc + 25;

%% ====================== FIGURE ================================
fig = figure( ...
    'Name','SIH26058 3-Sensor Adaptive LFM Demonstration', ...
    'NumberTitle','off', ...
    'Color','w', ...
    'Position',[100 70 1250 820]);

tl = tiledlayout(fig,2,1, ...
    'TileSpacing','compact', ...
    'Padding','compact');

%% ====================== TIME DOMAIN ================================
ax1 = nexttile(tl);

x = zeros(1,N);

hTime = plot(ax1, t*1e3, x, 'LineWidth',1.1);

grid(ax1,'on');
box(ax1,'on');

xlabel(ax1,'Time (ms)');
ylabel(ax1,'Amplitude');

title(ax1,'Adaptive LFM - Time Domain');

xlim(ax1,[0 Tchirp*1e3]);
ylim(ax1,[-1.1 1.1]);

%% ====================== FREQUENCY DOMAIN ===========================
ax2 = nexttile(tl);

f = (0:floor(N/2))*Fs/N;

hSpec = plot(ax2, ...
    f/1e3, ...
    zeros(size(f)), ...
    'LineWidth',1.3);

hold(ax2,'on');
grid(ax2,'on');
box(ax2,'on');

xlabel(ax2,'Frequency (kHz)');
ylabel(ax2,'Normalized Magnitude');

title(ax2,'Adaptive LFM - Spectrum');

% FIXED AXIS
xlim(ax2,[0 550]);
ylim(ax2,[0 1.05]);

%% ====================== C1-C5 REFERENCE BANDS ======================
for k = 1:length(modeNames)

    xline(ax2,bandLow(k),'--', ...
        'Alpha',0.25, ...
        'HandleVisibility','off');

    xline(ax2,bandHigh(k),'--', ...
        'Alpha',0.25, ...
        'HandleVisibility','off');

    xline(ax2,modeFc(k),':', ...
        'Alpha',0.25, ...
        'HandleVisibility','off');

    text(ax2, ...
        modeFc(k), ...
        0.95, ...
        sprintf('%s\n%d kHz',modeNames(k),modeFc(k)), ...
        'HorizontalAlignment','center', ...
        'VerticalAlignment','top', ...
        'FontWeight','bold', ...
        'FontSize',9);
end

%% ====================== ACTIVE BAND ================================
activeLow = fc_kHz - BW_kHz/2;
activeHigh = fc_kHz + BW_kHz/2;

hActiveLow = xline(ax2,activeLow,'-', ...
    'LineWidth',2.5);

hActiveHigh = xline(ax2,activeHigh,'-', ...
    'LineWidth',2.5);

hFcLine = xline(ax2,fc_kHz,'-', ...
    'LineWidth',3);

activeY = 0.78;

hActiveBand = plot(ax2, ...
    [activeLow activeHigh], ...
    [activeY activeY], ...
    'LineWidth',5);

hActiveLeftMarker = plot(ax2, ...
    activeLow,activeY,'o', ...
    'MarkerSize',6, ...
    'LineWidth',1.5);

hActiveRightMarker = plot(ax2, ...
    activeHigh,activeY,'o', ...
    'MarkerSize',6, ...
    'LineWidth',1.5);

%% ====================== STATUS AREA ===============================
hStatus = annotation(fig,'textbox', ...
    [0.03 0.005 0.94 0.06], ...
    'String','', ...
    'EdgeColor','none', ...
    'HorizontalAlignment','center', ...
    'FontSize',11, ...
    'FontWeight','bold');

sgtitle(tl, ...
    'SIH26058: 3-Sensor Adaptive Center Frequency + CONSTANT 50 kHz Bandwidth', ...
    'FontSize',14, ...
    'FontWeight','bold');

%% ====================== INITIAL PLOT ===============================
[x,freq,mag] = makeLFM( ...
    fc_kHz, ...
    BW_kHz, ...
    t, ...
    Fs, ...
    N, ...
    Tchirp);

set(hTime,'YData',x);

set(hSpec, ...
    'XData',freq/1e3, ...
    'YData',mag);

updateActiveBand( ...
    hActiveLow, ...
    hActiveHigh, ...
    hFcLine, ...
    hActiveBand, ...
    hActiveLeftMarker, ...
    hActiveRightMarker, ...
    fc_kHz, ...
    BW_kHz, ...
    activeY);

updateStatus( ...
    hStatus, ...
    Tenv, ...
    Depth_m, ...
    Range_cm, ...
    Mode, ...
    fc_kHz, ...
    BW_kHz);

drawnow;

%% ====================== MAIN LOOP ================================
disp(' ');
disp('SIH26058 3-sensor MATLAB visualizer is running.');
disp('Input 1 = Temperature POT');
disp('Input 2 = Depth POT');
disp('Input 3 = HC-SR04 target range');
disp('Frequency axis is FIXED at 0-550 kHz.');
disp('Close the figure window to stop.');
disp(' ');

while isvalid(fig)

    newestLine = "";

    % Keep only the newest valid DATA packet
    while s.NumBytesAvailable > 0

        line = strtrim(readline(s));

        if startsWith(line,"DATA,")
            newestLine = line;
        end
    end

    %% --------------------- PARSE DATA ------------------------------
    if strlength(newestLine) > 0

        parts = split(newestLine,",");

        % DATA,T,Depth,Range,fc,BW,Mode
        if numel(parts) == 7

            newT       = str2double(parts(2));
            newDepth   = str2double(parts(3));
            newRange   = str2double(parts(4));
            newFc      = str2double(parts(5));
            newBW      = str2double(parts(6));
            newMode    = string(parts(7));

            if all(isfinite([ ...
                    newT, ...
                    newDepth, ...
                    newRange, ...
                    newFc, ...
                    newBW]))

                Tenv = newT;
                Depth_m = newDepth;
                Range_cm = newRange;

                fc_kHz = newFc;
                BW_kHz = newBW;
                Mode = newMode;

                %% ---------------- GENERATE LFM -------------------
                [x,freq,mag] = makeLFM( ...
                    fc_kHz, ...
                    BW_kHz, ...
                    t, ...
                    Fs, ...
                    N, ...
                    Tchirp);

                %% ---------------- TIME DOMAIN --------------------
                set(hTime,'YData',x);

                %% ---------------- SPECTRUM -----------------------
                set(hSpec, ...
                    'XData',freq/1e3, ...
                    'YData',mag);

                % Do NOT change xlim.
                % It permanently stays 0-550 kHz.

                %% ---------------- ACTIVE BAND --------------------
                updateActiveBand( ...
                    hActiveLow, ...
                    hActiveHigh, ...
                    hFcLine, ...
                    hActiveBand, ...
                    hActiveLeftMarker, ...
                    hActiveRightMarker, ...
                    fc_kHz, ...
                    BW_kHz, ...
                    activeY);

                %% ---------------- TITLES -------------------------
                activeLow = fc_kHz - BW_kHz/2;
                activeHigh = fc_kHz + BW_kHz/2;

                title(ax1,sprintf( ...
                    'Adaptive LFM - Time Domain | fc = %.1f kHz | BW = %.1f kHz', ...
                    fc_kHz, ...
                    BW_kHz));

                title(ax2,sprintf( ...
                    'Spectrum | Mode = %s | Active Band = %.1f-%.1f kHz', ...
                    Mode, ...
                    activeLow, ...
                    activeHigh));

                %% ---------------- STATUS -------------------------
                updateStatus( ...
                    hStatus, ...
                    Tenv, ...
                    Depth_m, ...
                    Range_cm, ...
                    Mode, ...
                    fc_kHz, ...
                    BW_kHz);

                drawnow limitrate;
            end
        end
    end

    pause(0.05);
end

%% ====================== CLEANUP ===================================
clear s;

disp('MATLAB visualizer stopped.');

%% ====================== LOCAL FUNCTION: LFM =======================
function [x,freq,mag] = makeLFM( ...
    fc_kHz, ...
    BW_kHz, ...
    t, ...
    Fs, ...
    N, ...
    Tchirp)

    fc = fc_kHz*1e3;
    BW = BW_kHz*1e3;

    fStart = fc - BW/2;

    chirpRate = BW/Tchirp;

    phase = 2*pi*( ...
        fStart*t + ...
        0.5*chirpRate*t.^2);

    x = sin(phase);

    % Manual Hann window
    window = 0.5 - 0.5*cos( ...
        2*pi*(0:N-1)/(N-1));

    X = fft(x .* window);

    X = X(1:floor(N/2)+1);

    freq = (0:floor(N/2))*Fs/N;

    mag = abs(X);

    if max(mag) > 0
        mag = mag/max(mag);
    end
end

%% ====================== ACTIVE BAND FUNCTION ======================
function updateActiveBand( ...
    hActiveLow, ...
    hActiveHigh, ...
    hFcLine, ...
    hActiveBand, ...
    hActiveLeftMarker, ...
    hActiveRightMarker, ...
    fc_kHz, ...
    BW_kHz, ...
    activeY)

    activeLow = fc_kHz - BW_kHz/2;
    activeHigh = fc_kHz + BW_kHz/2;

    set(hActiveLow,'Value',activeLow);
    set(hActiveHigh,'Value',activeHigh);
    set(hFcLine,'Value',fc_kHz);

    set(hActiveBand, ...
        'XData',[activeLow activeHigh], ...
        'YData',[activeY activeY]);

    set(hActiveLeftMarker, ...
        'XData',activeLow, ...
        'YData',activeY);

    set(hActiveRightMarker, ...
        'XData',activeHigh, ...
        'YData',activeY);
end

%% ====================== STATUS FUNCTION ===========================
function updateStatus( ...
    hStatus, ...
    Tenv, ...
    Depth_m, ...
    Range_cm, ...
    Mode, ...
    fc_kHz, ...
    BW_kHz)

    lowFreq = fc_kHz - BW_kHz/2;
    highFreq = fc_kHz + BW_kHz/2;

    hStatus.String = sprintf( ...
        'Temperature = %.2f ^oC   |   Depth = %.1f m   |   HC-SR04 Range = %.1f cm   |   Mode = %s   |   fc = %.1f kHz   |   BW = %.1f kHz   |   Active band = %.1f-%.1f kHz', ...
        Tenv, ...
        Depth_m, ...
        Range_cm, ...
        Mode, ...
        fc_kHz, ...
        BW_kHz, ...
        lowFreq, ...
        highFreq);
end
